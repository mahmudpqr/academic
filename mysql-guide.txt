MODIFY column_name dataType AUTO_INCREMENT;

AUTO_INCREMENT = nextValue;

MODIFY column_name dataType;

data types: [UNSIGNED] integers
constraints: PRIMARY KEY, UNIQUE NOT NULL
gaps: delete, failed insert, transaction rollback
counter unchanged: inserting null or value < max, setting next value < max
counter changed: inserting value > max, setting next value > max

    [ON UPDATE cascadingAction]
