==================== foreign key ====================

CREATE TABLE table_name (
    column_name dataType REFERENCES parentTable(parentColumn)
);

CREATE TABLE table_name
(
    column1 dataType,
    column2 dataType,

    [CONSTRAINT constraint_name] FOREIGN KEY (column1, column2)
    REFERENCES parentTable(parentColumn1, parentColumn2)
    [cascading actions]
);

ALTER TABLE table_name
ADD [CONSTRAINT constraint_name] FOREIGN KEY (column1, column2)
REFERENCES parentTable(parentColumn1, parentColumn2);
[cascading actions]

ALTER TABLE table_name
DROP FOREIGN KEY constraint_name;

cascading actions:
ON DELETE/UPDATE CASCADE / RESTRICT / NO ACTION / SET NULL
