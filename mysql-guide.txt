==================== foreign key ====================

CREATE TABLE tableName (
    columnName dataType REFERENCES parentTable(parentColumn)
);

CREATE TABLE tableName
(
    column1 dataType,
    column2 dataType,

    [CONSTRAINT constraintName] FOREIGN KEY (column1, column2)
    REFERENCES parentTable(parentColumn1, parentColumn2)
    [cascading actions]
);

ALTER TABLE tableName
ADD [CONSTRAINT constraintName] FOREIGN KEY (column1, column2)
REFERENCES parentTable(parentColumn1, parentColumn2);
[cascading actions]

ALTER TABLE tableName
DROP FOREIGN KEY constraintName;

cascading actions:
ON DELETE/UPDATE CASCADE / RESTRICT / NO ACTION / SET NULL
